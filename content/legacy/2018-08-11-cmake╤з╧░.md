### 最简单的 cmakelist {#最简单的-cmakelist}

```plain
PROJECT (HELLO)
# <projectname>_BINARY_DIR  <projectname>_SOURCE_DIR
SET(SRC_LIST main.c)
MESSAGE(STATUS "This is BINARY dir " ${HELLO_BINARY_DIR})
MESSAGE(STATUS "This is SOURCE dir "${HELLO_SOURCE_DIR})
ADD_EXECUTABLE(hello ${SRC_LIST})
```

### 内部编译和外部编译 {#内部编译和外部编译}

通过外部编译进行工程构建，PROJECT\_SOURCE\_DIR仍然指代工程路径，即/backup/cmake/t1

而PROJECT\_BINARY\_DIR则指代编译路径，即/backup/cmake/t1/build

### 看着正规一点 (工程化) {#看着正规一点-工程化}

#### 编译库 {#编译库}

结构

```plain
.
|-- build
|-- CMakeLists.txt
|-- lib
|   |-- libhello.a
|   `-- libhello.so
`-- libsrc
    |-- CMakeLists.txt
    |-- hello.c
    `-- hello.h
```

src/cmakelists:

```plain
SET(LIBHELLO_SRC hello.c)
ADD_LIBRARY(hello SHARED ${LIBHELLO_SRC})
ADD_LIBRARY(hellost STATIC ${LIBHELLO_SRC})
# set name  为了生成 同名的 静态库和动态库
SET_TARGET_PROPERTIES(hellost PROPERTIES OUTPUT_NAME "hello")
SET(LIBRARY_OUTPUT_PATH ${PROJECT_SOURCE_DIR}/lib)
```

hello.c

```plain
#include "hello.h"
void HelloFunc()
{
    printf("Hello World\n");
}
```

hello.h

```plain
#ifndef HELLO_H
#define HELLO_H
#include <stdio.h>
void HelloFunc();
#endif
```

#### 使用头文件和库 {#使用头文件和库}

结构:

```plain
.
|-- 3rdparty
|   `-- hello.h
|-- bin
|   `-- main
|-- build
|-- CMakeLists.txt
|-- lib
|   |-- libhello.a
|   `-- libhello.so
`-- src
    |-- CMakeLists.txt
    `-- main.c
```

src/cmakelists.txt

```plain
# 头文件
INCLUDE_DIRECTORIES(${PROJECT_SOURCE_DIR}/3rdparty)
# 库文件 路径
LINK_DIRECTORIES(${PROJECT_SOURCE_DIR}/lib)
ADD_EXECUTABLE(main main.c)
# 链接的库
TARGET_LINK_LIBRARIES(main libhello.so)
# 指定生成 路径
SET(EXECUTABLE_OUTPUT_PATH ${PROJECT_SOURCE_DIR}/bin)
```

### 主要指令 {#主要指令}

```plain
# 加入 子路径
ADD_SUBDIRECTORY(source_dir [binary_dir] [EXCLUDE_FROM_ALL])

# 通过SET指令重新定义EXECUTABLE_OUTPUT_PATH 
SET(EXECUTABLE_OUTPUT_PATH ${PROJECT_BINARY_DIR}/bin)
SET(LIBRARY_OUTPUT_PATH ${PROJECT_BINARY_DIR}/lib)
# 在哪里ADD_EXECUTABLE或ADD_LIBRARY，如果需要改变目标存放路径，就在哪里加入上述的定义

# 编译 静态库
ADD_LIBRARY(hello STATIC ${LIBHELLO_SRC})

# 头文件 搜索路径
INCLUDE_DIRECTORIES(/usr/include/hello)

# 共享库 搜索路径
LINK_DIRECTORIES(directory1 directory2 ...)

# 添加需要链接的共享库
TARGET_LINK_LIBRARIES(target library1
    <debug | optimized> library2
    ...)
```
